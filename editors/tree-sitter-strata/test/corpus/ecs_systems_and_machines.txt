timeline Slash @rate(60_hz) {
    window active = 2_f..6_f;
    at 2_f => trigger(self, Hit { dmg: 10.0 });
}

task Dash(speed: mps = 20.0_mps)
    on (vel: &mut Vel2D.dxy)
    .with<Alive>()
    using (clk: &res SimClock)
{
    abort_if speed.raw() <= 0.0 { return; }
    race {
        branch {
            yield_tick;
        }
        branch {
            wait 100ms;
            wait_until true;
        }
    }
}

fsm GuardFsm(clk: &res SimClock) @require(Pos2D) {
    initial Patrol { step: 0 };
    state Patrol {
        step: u32 = 0,
        on_enter { let a = 1; }
        on_tick { let b = 2; }
        on_exit { let c = 3; }
        to Chase when clk.dt > 0.0_s;
    }
}

@par(chunk = 1024)
sys step_world(clk: &res SimClock) effects(writes: [Pos3D]) {
    uniform let dt = clk.dt;
    for row in query(hp: &mut? Hull.hp, opt: ?&Shield, caps: &Hull::caps) {
        row.hp += 1.0;
    }
}

obs on_hit(trg: &Trigger<HitEvent>) {
    despawn_all<With<Temp>>();
    remove<Shield>(trg.target);
    insert @if_new(trg.target, Stunned {});
}

migrate Hull from 1.0.0 to 2.0.0(old: &HullV1) -> Hull {
    Hull { hp: old.hp as f32 }
}

assert_layout(Hull::hot, size == 16);
assert_par(step_world, audit_world);

schedule FixedUpdate run_if(res<SimClock>().dt > 0.0_s) {
    seq Core {
        par Jobs { step_world }
    }
}

sim_test "ticks_cleanly" {
    step_fixed(1);
}

net_sim_test "syncs"(latency: 40ms, clients: 2) {
    let q = client_query(0, h: &Hull).on(Hero);
}
