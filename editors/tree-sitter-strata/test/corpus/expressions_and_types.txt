fn use_contextual_words(
    uniform: u32,
    abort_if: u32,
    race: u32,
    wait: u32,
    wait_until: u32,
    yield_tick: u32,
    atomic: u32,

    insert: u32,
    remove: u32,
    query: u32,
    query_one: u32,
    server_query: u32,
    client_query: u32,
) -> u32 {
    let mut acc = uniform + abort_if + race + wait + wait_until + yield_tick + atomic + spawn;
    acc += spawn_batch + despawn_all + insert + remove + query + query_one + server_query + client_query;
    let _ = wait(acc);
    return acc;
}
